% --- SETTINGS ---
\newcommand{\CurrentModel}{2, 1n, 0nn, 0---0, ee, me, mm} % Values: Comma-separated list
\newcommand{\DisplayMode}{Both}  % Values: Curve, Graph, or Both

% --- HELPER COMMANDS FOR DUAL GRAPHS ---
\newcommand{\nd}[3]{\node[shape=circle,draw=black,scale=1.0,line width=0.8pt] (#1) at (#2) {#3};}
\newcommand{\lp}[3]{\path (#1) edge [loop,min distance=1.5cm,in=#2,out=#3,line width=0.8pt] node {} (#1);}
\newcommand{\ed}[2]{\draw[line width=0.8pt] (#1) -- (#2);}
% Command for curved edges
\newcommand{\edb}[3]{\draw[line width=0.8pt] (#1) to[bend #3] (#2);} 

% --- WRAPPER COMMANDS ---
\newcommand{\HobbyCurve}[2][]{%
  \csname curve:#2\endcsname[#1]%
}
\newcommand{\DualGraph}[2][]{%
  \csname graph:#2\endcsname[#1]%
}

% ==============================================================================
% --- CURVE DEFINITIONS ---
% ==============================================================================
\expandafter\newcommand\csname curve:2\endcsname[1][]{%
\begin{tikzpicture}[scale=0.3, #1]
  \useasboundingbox (-0.2,-0.7) rectangle (3.7,1.2);
  \clip (-0.2,-0.7) rectangle (3.7,1.2);
  \draw[line width = 4] (0, 0) to (3.5, 0);
\end{tikzpicture}%
}

\expandafter\newcommand\csname curve:1n\endcsname[1][]{%
\begin{tikzpicture}[scale=0.2, #1]
  \useasboundingbox (-0.2,-0.2) rectangle (4.2,3.2);
  \clip (-0.2,-0.2) rectangle (4.2,3.2);
  \draw[line width = 2] (0, 0) to[quick curve through={(2, 0.5) (3, 1.5) (2, 3) (1, 1.5) (2, 0.5)}] (4, 0);
\end{tikzpicture}%
}

\expandafter\newcommand\csname curve:0nn\endcsname[1][]{%
\begin{tikzpicture}[scale=0.2, xscale=0.8, #1]
  \useasboundingbox (-0.2,-0.2) rectangle (8.2,3.2);
  \clip (-0.2,-0.2) rectangle (8.2,3.2);
  \draw (0, 0) to[quick curve through={(2, 0.5) (3, 1.5) (2, 3) (1, 1.5) (2, 0.5) (4, 0) (6, 0.5) (7, 1.5) (6, 3) (5, 1.5) (6, 0.5)}] (8, 0);
\end{tikzpicture}%
}

\expandafter\newcommand\csname curve:ee\endcsname[1][]{%

\newif\ifRenderThisModel
\newcommand{\RenderModel}[3]{%
  \global\RenderThisModelfalse
  \foreach \model in \CurrentModel {%
    \ifthenelse{\equal{\model}{#1}}{\global\RenderThisModeltrue}{}%
  }%
  \ifRenderThisModel
    \ifthenelse{\equal{\DisplayMode}{Curve} \OR \equal{\DisplayMode}{Both}}{#2}{}%
    \ifthenelse{\equal{\DisplayMode}{Graph} \OR \equal{\DisplayMode}{Both}}{#3}{}%
  \fi
}
